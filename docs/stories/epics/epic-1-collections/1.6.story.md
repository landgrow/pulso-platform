## Descrição

Adicionar versionamento explícito às coleções: cada update cria uma nova revision (não muta a anterior), UI mostra timeline com diff visual, e permite rollback.

## Notas Técnicas

### Schema `colecao_revisions`

```sql
CREATE TABLE colecao_revisions (
  id uuid PK DEFAULT gen_random_uuid(),
  colecao_id uuid NOT NULL REFERENCES colecoes(id) ON DELETE CASCADE,
  rev_number int NOT NULL,
  payload jsonb NOT NULL,
  metadata jsonb NOT NULL DEFAULT '{}'::jsonb,
  change_summary jsonb NOT NULL DEFAULT '[]'::jsonb,  -- ["financeiro.faturamento", "comercial.clientes_ativos"]
  created_by uuid REFERENCES profiles(id) ON DELETE SET NULL,
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (colecao_id, rev_number)
);

CREATE INDEX idx_revisions_colecao ON colecao_revisions(colecao_id, rev_number DESC);

ALTER TABLE colecao_revisions ENABLE ROW LEVEL SECURITY;
ALTER TABLE colecao_revisions FORCE ROW LEVEL SECURITY;

-- RLS via propagação do periodo_id
CREATE POLICY revisions_select ON colecao_revisions
  FOR SELECT USING (
    EXISTS (
      SELECT 1 FROM colecoes c
      JOIN periodos_dados p ON p.id = c.periodo_id
      JOIN clientes cl ON cl.id = p.cliente_id
      WHERE c.id = colecao_revisions.colecao_id
        AND public.can_access_org(cl.org_id)
    )
  );
```

### Trigger de versionamento

```sql
CREATE OR REPLACE FUNCTION public.archive_colecao_revision()
RETURNS trigger
LANGUAGE plpgsql
AS $$
DECLARE
  v_rev int;
  v_changes jsonb;
BEGIN
  -- Próximo rev_number
  SELECT COALESCE(MAX(rev_number), 0) + 1
  INTO v_rev
  FROM colecao_revisions
  WHERE colecao_id = OLD.id;

  -- Change summary: chaves que mudaram
  SELECT jsonb_agg(key)
  INTO v_changes
  FROM jsonb_object_keys(OLD.payload) AS key
  WHERE OLD.payload->key IS DISTINCT FROM NEW.payload->key;

  INSERT INTO colecao_revisions (
    colecao_id, rev_number, payload, metadata, change_summary, created_by
  ) VALUES (
    OLD.id, v_rev, OLD.payload, OLD.metadata, COALESCE(v_changes, '[]'::jsonb), auth.uid()
  );

  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_colecoes_archive ON colecoes;
CREATE TRIGGER trg_colecoes_archive
  BEFORE UPDATE ON colecoes
  FOR EACH ROW EXECUTE FUNCTION public.archive_colecao_revision();
```

### Server action `restoreColecao`

```typescript
// src/app/actions/colecoes.ts (adição)
const restoreSchema = z.object({
  colecaoId: z.string().uuid(),
  revNumber: z.number().int().positive(),
});

export async function restoreColecao(input: unknown): Promise<Result<{ newRev: number }>> {
  const parsed = restoreSchema.parse(input);
  const supabase = await createClient();
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return { success: false, error: "Not authenticated", code: "PERMISSION_DENIED" };

  // Verifica período não-fechado
  const { data: colecao } = await supabase
    .from("colecoes")
    .select("id, periodo_id, periodos_dados(status)")
    .eq("id", parsed.colecaoId)
    .single();

  if (!colecao) return { success: false, error: "Não encontrada", code: "NOT_FOUND" };
  // @ts-expect-error - join
  if (colecao.periodos_dados?.status === "fechado") {
    return { success: false, error: "Período fechado", code: "PERIOD_CLOSED" };
  }

  // Busca revision alvo
  const { data: rev } = await supabase
    .from("colecao_revisions")
    .select("payload, metadata")
    .eq("colecao_id", parsed.colecaoId)
    .eq("rev_number", parsed.revNumber)
    .single();

  if (!rev) return { success: false, error: "Revisão não encontrada", code: "NOT_FOUND" };

  // Aplica (trigger cria nova revision automaticamente)
  const { error } = await supabase
    .from("colecoes")
    .update({
      payload: rev.payload,
      metadata: rev.metadata,
      updated_at: new Date().toISOString(),
    })
    .eq("id", parsed.colecaoId);

  if (error) return { success: false, error: error.message, code: "INTERNAL_ERROR" };

  await logAudit(supabase, /* org_id */, "restore", "colecao", parsed.colecaoId, {
    rev_number: parsed.revNumber,
  });

  return { success: true, data: { newRev: parsed.revNumber + 1 } };
}
```

### Diff viewer

```typescript
// src/lib/diff.ts
import * as jsondiffpatch from "jsondiffpatch";

const differ = jsondiffpatch.create({
  objectHash: (obj: unknown) => JSON.stringify(obj),
  textDiff: { minLength: 60 },
});

export type DiffResult = {
  added: string[];
  removed: string[];
  changed: string[];
};

export function diffPayloads(
  before: Record<string, unknown>,
  after: Record<string, unknown>,
): DiffResult {
  const delta = differ.diff(before, after);
  if (!delta) return { added: [], removed: [], changed: [] };

  const added: string[] = [];
  const removed: string[] = [];
  const changed: string[] = [];

  for (const [key, value] of Object.entries(delta)) {
    if (Array.isArray(value)) {
      if (value.length === 1) added.push(key);
      else if (value.length === 2) changed.push(key);
      else if (value.length === 3 && value[2] === 0) removed.push(key);
      else changed.push(key);
    } else if (typeof value === "object" && value !== null) {
      changed.push(key); // nested change
    }
  }

  return { added, removed, changed };
}
```

## Change Log

### Implementação (05/09/2026)

**Decisão de design — trigger em vez de reescrever `upsertColecao`:** a AC-2 original pedia para modificar `upsertColecao` para "criar nova revision ao invés de mutar". Em vez de duplicar essa lógica em `upsertColecao` (colecoes.ts) e `updateTextoLivre` (texto.ts) — e em qualquer futuro update de coleção —, o versionamento foi implementado como um trigger `BEFORE UPDATE ON colecoes` (`trg_colecoes_archive`). Toda vez que qualquer código já existente faz um `.update()` em `colecoes` (upsert de formulário, edição de texto livre, restore), o Postgres arquiva a versão anterior automaticamente, antes de aplicar o novo valor — sem precisar tocar nenhuma dessas actions. Isso também garante a atomicidade pedida na AC-2 por construção (não existe um caminho onde o UPDATE aconteça sem o INSERT de arquivo rodar antes, no mesmo statement).

**Bug corrigido no trigger de exemplo da própria story:** o SQL de exemplo em "Notas Técnicas" calculava `change_summary` iterando só `jsonb_object_keys(OLD.payload)` — então um campo que só existe em `NEW.payload` (um campo novo sendo adicionado) nunca aparecia no `change_summary`. Corrigido fazendo union das chaves de `OLD` e `NEW`.

**Adição fora do escopo original — coluna `colecoes.updated_by`:** o exemplo de trigger da story grava `created_by` da revision como `auth.uid()` de quem está fazendo o UPDATE atual — ou seja, "quem substituiu esta versão", não "quem escreveu esta versão". Isso inverteria a autoria mostrada na timeline. Corrigido adicionando `colecoes.updated_by` (não existia na migration 0005) — o trigger agora seta `NEW.updated_by = auth.uid()` a cada UPDATE e arquiva `COALESCE(OLD.updated_by, OLD.created_by)` como autor da revision, o que reflete corretamente quem escreveu aquela versão.

**`listColecaoRevisions` como arquivo novo (`revisions.ts`), não dentro de `colecoes.ts`:** a AC-3 precisa carregar todas as revisions + a versão atual "virtual" de cada coleção do período (a versão vigente só ganha uma linha em `colecao_revisions` na _próxima_ edição). Isso é uma leitura/agregação para exibição, não uma operação de CRUD de coleção — separado em `src/app/actions/revisions.ts` para não inchar `colecoes.ts`.

**Lib de diff:** usada a lib `diff` (jsdiff) em vez de `jsondiffpatch` (sugerida como alternativa na própria story) — mais simples para os dois casos reais (diff de linha para texto, diff raso de campos para formulário) sem precisar da árvore de diff genérica do jsondiffpatch. Instalada a v9 diretamente — a v8.0.2 (e toda a faixa 6.x-8.0.2) tem uma vulnerabilidade de DoS conhecida (GHSA-73rr-hh4g-fpgx) em `parsePatch`/`applyPatch`, funções que este projeto nem usa (só `diffLines`), mas não fazia sentido instalar uma versão vulnerável por padrão.

**Verificação da migração — limitação do ambiente:** as migrations `0007` e `0008` foram escritas replicando cuidadosamente os padrões já em uso no projeto (`SECURITY DEFINER SET search_path = public` do `log_audit()` em 0003; a policy de INSERT do `audit_log` em 0002 como modelo pra `colecao_revisions`; o alias `periodos_dados!inner` já corrigido na Story 1.5), mas **nunca foram executadas contra um Postgres real** — este ambiente de desenvolvimento não tem `docker` instalado (o projeto tem um `docker-compose.yml` com Supabase self-hosted completo, mas o binário não está disponível aqui) nem `DATABASE_URL` configurado em `.env.local` (só `SUPABASE_SERVICE_ROLE_KEY`). Isso significa que o trigger, a RLS de `colecao_revisions` e as funções de cleanup foram revisados linha a linha mas não têm confirmação de que rodam sem erro de sintaxe/sequência em cima das migrations 0001-0006 já aplicadas. Aberta uma tarefa separada para alguém com acesso a Docker/Postgres real aplicar as 8 migrations em sequência e validar o trigger fim-a-fim antes de considerar esta story PASS de verdade — até lá, o gate fica em CONCERNS.

**Testes:** os pontos de AC-6 que dependem de Postgres real (trigger, rollback) não têm teste automatizado nesta entrega, pela mesma razão acima — e são exatamente os únicos tipos de teste que a convenção deste projeto (Zod sem mockar Supabase) não cobre. `diffFields`/`diffText`/`changedKeys` são funções puras e têm 12 testes cobrindo added/removed/changed/idêntico/múltiplas mudanças/linha adicionada/removida/idêntica. `restoreColecaoSchema` e `listRevisionsSchema` têm 6 testes de validação (mesmo padrão de `periods.test.ts`/`evidencias.test.ts`).

**Gates:** typecheck ✅, lint ✅, build ✅, test ✅ (159/159 — 18 novos), e2e ✅ (mesmo escopo de autenticação obrigatória das stories anteriores; os 4 testes pré-existentes de `public-pages.spec.ts` continuam falhando pela mesma causa já registrada desde a Story 1.3, e firefox continua sem executável instalado).
