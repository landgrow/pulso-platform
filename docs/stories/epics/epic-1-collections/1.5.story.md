## Notas Técnicas

### Wrapper Claude API

```typescript
// src/lib/claude.ts
import Anthropic from "@anthropic-ai/sdk";

let _client: Anthropic | null = null;

export function getAnthropicClient(): Anthropic {
  if (_client) return _client;
  _client = new Anthropic({
    apiKey: process.env.ANTHROPIC_API_KEY!,
  });
  return _client;
}

export async function transcribeAudio(
  audioBase64: string,
  mimeType: string,
): Promise<string> {
  const client = getAnthropicClient();
  const response = await client.messages.create({
    model: "claude-sonnet-4-5",
    max_tokens: 4096,
    messages: [
      {
        role: "user",
        content: [
          {
            type: "audio",
            source: {
              type: "base64",
              media_type: mimeType as "audio/mp3" | "audio/wav" | "audio/m4a",
              data: audioBase64,
            },
          },
          {
            type: "text",
            text: "Transcreva o áudio a seguir em português brasileiro. Mantenha pontuação e números por extenso. Use marcadores [00:30] para indicar tempo se relevante.",
          },
        ],
      },
    ],
  });

  const block = response.content[0];
  if (block.type !== "text") throw new Error("Resposta inesperada da Claude");
  return block.text;
}
```

### Server action `transcribeAudio`

```typescript
// src/app/actions/transcricao.ts
"use server";

import { transcribeAudio as claudeTranscribe } from "@/lib/claude";
import { createClient } from "@/lib/supabase/server";

export async function transcribeAudio(
  evidenciaId: string,
): Promise<Result<{ text: string }>> {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user)
    return {
      success: false,
      error: "Not authenticated",
      code: "PERMISSION_DENIED",
    };

  // Busca evidência
  const { data: ev } = await supabase
    .from("evidencias")
    .select("id, storage_path, mime_type, transcricao_status, transcricao")
    .eq("id", evidenciaId)
    .single();

  if (!ev)
    return {
      success: false,
      error: "Evidência não encontrada",
      code: "NOT_FOUND",
    };
  if (ev.transcricao_status === "completed" && ev.transcricao) {
    return { success: true, data: { text: ev.transcricao } };
  }

  // Marca como processing
  await supabase
    .from("evidencias")
    .update({ transcricao_status: "processing" })
    .eq("id", evidenciaId);

  try {
    // Baixa do storage
    const { data: fileData, error: downloadError } = await supabase.storage
      .from("evidencias")
      .download(ev.storage_path);
    if (downloadError) throw downloadError;

    const buffer = Buffer.from(await fileData.arrayBuffer());
    const base64 = buffer.toString("base64");

    // Chama Claude
    const text = await Promise.race([
      claudeTranscribe(base64, ev.mime_type),
      new Promise<never>((_, reject) =>
        setTimeout(() => reject(new Error("Timeout")), 120_000),
      ),
    ]);

    // Salva resultado
    await supabase
      .from("evidencias")
      .update({
        transcricao: text,
        transcricao_status: "completed",
      })
      .eq("id", evidenciaId);

    return { success: true, data: { text } };
  } catch (err) {
    await supabase
      .from("evidencias")
      .update({
        transcricao_status: "failed",
        transcricao: `Erro: ${err instanceof Error ? err.message : "desconhecido"}`,
      })
      .eq("id", evidenciaId);

    return {
      success: false,
      error: "Falha na transcrição",
      code: "INTERNAL_ERROR",
    };
  }
}
```

// src/lib/validations/audio.ts
import { z } from "zod";

const ACCEPTED_AUDIO_MIMES = [
  "audio/mp3",
  "audio/mpeg",
  "audio/wav",
  "audio/x-wav",
  "audio/m4a",
  "audio/mp4",
];
const MAX_AUDIO_SIZE = 50 * 1024 * 1024; // 50MB
const MAX_DURATION_SEC = 60 * 60; // 60min

export const audioUploadSchema = z.object({
  periodoId: z.string().uuid(),
  storagePath: z.string(),
  nomeOriginal: z.string().min(1).max(255),
  mimeType: z.string().refine((m) => ACCEPTED_AUDIO_MIMES.includes(m), {
    message: "Tipo de áudio não suportado",
  }),
  tamanhoBytes: z.number().int().positive().max(MAX_AUDIO_SIZE),
});

## Change Log

### Retomada (05/09/2026)

Story retomada após sessão anterior (Cursor) ficar sem créditos antes de qualquer arquivo desta story existir no disco — ao contrário das Stories 1.3/1.4, aqui não havia código prévio, só o texto da story.

**Achado crítico antes de implementar áudio:** a seção "Notas Técnicas" desta story (código-exemplo de `src/lib/claude.ts` e `transcribeAudio`) assume que a Claude Messages API aceita um content block `{ type: "audio", source: {...} }`. Isso **não existe** na API real — confirmado via a skill `claude-api` (docs oficiais empacotadas) e um WebFetch ao vivo de `platform.claude.com/docs/en/build-with-claude/vision`. Os únicos content blocks suportados são `text`, `image` e `document` (PDF). Esse design provavelmente veio de uma alucinação de quem escreveu a story original (Cursor) — nenhuma versão da API Anthropic jamais teve capacidade nativa de transcrição de áudio.

**Decisão: áudio adiado.** Reportado o achado à Nayara, que decidiu explicitamente: _"Adiar áudio, seguir só com texto livre agora"_. Portanto:

- AC-3, AC-4, e as partes de áudio de AC-5/AC-6 ficam **adiadas**, não "pendentes" por falha — é uma decisão de escopo, documentada aqui para não ser confundida com trabalho esquecido.
- Antes de retomar o áudio, será preciso decidir um provedor de ASR real (Whisper da OpenAI, Deepgram, AssemblyAI, Google Speech-to-Text ou AWS Transcribe) e desenhar a integração em cima dele — não da Claude API.
- A tabela `evidencias` já tem as colunas `transcricao` e `transcricao_status` prontas (migração 0005) e o bucket Storage já aceita MIME types de áudio (migração 0006) — nenhuma migração nova será necessária quando o áudio for retomado, só a lógica de chamada ao ASR escolhido.

**Implementado (texto livre):**

- `src/lib/validations/texto.ts` — schemas `createTextoLivreSchema`, `updateTextoLivreSchema`, `listTextoLivreSchema` e a constante `TEXTO_AREAS` (as mesmas 6 áreas do formulário estruturado + "Geral", conforme AC-1).
- `src/app/actions/texto.ts` — `createTextoLivre`, `updateTextoLivre`, `listTextoLivre`. Decisão de design: **não** reaproveitar `upsertColecao` (de `colecoes.ts`), porque seu upsert por `(periodo_id, tipo)` colapsaria múltiplas entradas de texto na mesma linha — incompatível com AC-1 ("lista de textos salvos"). `deleteColecao`/`discardColecao` (delete) foram reaproveitados de `colecoes.ts` normalmente, pois seu soft-delete por `id` é genérico o bastante.
- UI: `texto-form.tsx` (criação manual, sem auto-save — conforme AC-1), `texto-list.tsx` (edição inline + exclusão), `texto-page-client.tsx` (orquestra estado local entre form/lista), `page.tsx`.

**Bug pré-existente corrigido (bloqueava o delete de texto livre):** `deleteColecao` e `discardColecao`, em `src/app/actions/colecoes.ts`, usavam `.select("... periodo_dados!inner(...)")` — mas a tabela real se chama `periodos_dados` (com "s"), e o PostgREST resolve o embed pelo nome real da tabela-alvo. Como não existia relacionamento chamado `periodo_dados`, a query falhava e ambas as funções sempre retornavam "Coleção não encontrada" (`NOT_FOUND`), mesmo com um id válido. Esse bug já existia antes desta story (nenhum teste cobria `discardColecao`/`deleteColecao` — a convenção de testes deste projeto testa só schemas Zod, sem chamar o Supabase de verdade) e afetava também a Story 1.3, mas só virou bloqueante agora porque a Story 1.5 depende de `discardColecao` para excluir textos. Corrigido trocando o alias para `periodos_dados!inner` nas duas funções.

**Achado de segurança não corrigido aqui (fora de escopo, tarefa separada aberta):** as políticas RLS `colecoes_insert`/`colecoes_update` (migração `0005_collections_schema.sql`) checam só `can_access_org(org_id)`, sem excluir o papel `client_viewer` — ou seja, um `client_viewer` (que a AC-5 desta story e a UI tratam como somente-leitura) consegue inserir/atualizar uma `colecao` chamando o banco diretamente, contornando a checagem de `canEdit` da aplicação. Esse gap é pré-existente (afeta igualmente o `upsertColecao` da Story 1.3, não é algo introduzido aqui) e foi sinalizado como tarefa separada em vez de corrigido nesta entrega, por alterar uma policy RLS compartilhada entre stories já concluídas.
